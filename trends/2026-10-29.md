import numpy as np

def calculate_slope(points):
    """
    Calculates the slope of the best-fit line for a given set of points using linear regression.

    Args:
        points (list): A list of tuples, where each tuple represents a point (x, y).

    Returns:
        float: The slope of the best-fit line.
    """
    if len(points) < 2:
        return np.nan  # Not enough points to calculate a slope

    x_coords = np.array([p[0] for p in points])
    y_coords = np.array([p[1] for p in points])

    # Calculate the means of x and y
    mean_x = np.mean(x_coords)
    mean_y = np.mean(y_coords)

    # Calculate the numerator and denominator for the slope formula
    numerator = np.sum((x_coords - mean_x) * (y_coords - mean_y))
    denominator = np.sum((x_coords - mean_x)**2)

    # Avoid division by zero if all x-coordinates are the same
    if denominator == 0:
        return np.inf if np.any(y_coords != mean_y) else np.nan
    
    slope = numerator / denominator
    return slope
